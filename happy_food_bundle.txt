import 'package:shared_preferences/shared_preferences.dart';

/// Remembers which role the person chose as default on this phone.
class RoleStore {
  static const _key = 'default_role';

  static Future<String?> getDefault() async {
    try {
      return (await SharedPreferences.getInstance()).getString(_key);
    } catch (_) {
      return null;
    }
  }

  static Future<void> setDefault(String? role) async {
    try {
      final p = await SharedPreferences.getInstance();
      if (role == null) {
        await p.remove(_key);
      } else {
        await p.setString(_key, role);
      }
    } catch (_) {}
  }
}

import 'package:flutter/material.dart';
import 'package:geolocator/geolocator.dart';

import '../../main.dart';

/// Customer: pick dishes, place the order, then watch the radar.
class CustomerHome extends StatefulWidget {
  const CustomerHome({super.key});

  @override
  State<CustomerHome> createState() => _CustomerHomeState();
}

class _CustomerHomeState extends State<CustomerHome> {
  List<Map<String, dynamic>> _products = [];
  final Map<String, int> _cart = {};
  bool _loading = true;
  bool _placing = false;

  @override
  void initState() {
    super.initState();
    _load();
  }

  void _msg(String m) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(m)));
  }

  Future<void> _load() async {
    try {
      final rows = await supabase
          .from('products')
          .select('id, name, price')
          .eq('is_active', true)
          .order('name');
      if (!mounted) return;
      setState(() => _products = List<Map<String, dynamic>>.from(rows));
    } catch (e) {
      _msg('लोड नहीं हुआ: $e');
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  num get _total {
    num t = 0;
    for (final p in _products) {
      t += (p['price'] as num) * (_cart[p['id']] ?? 0);
    }
    return t;
  }

  Future<Position?> _position() async {
    if (!await Geolocator.isLocationServiceEnabled()) {
      _msg('फ़ोन की लोकेशन (GPS) चालू कीजिए');
      return null;
    }
    var p = await Geolocator.checkPermission();
    if (p == LocationPermission.denied) p = await Geolocator.requestPermission();
    if (p == LocationPermission.denied || p == LocationPermission.deniedForever) {
      _msg('लोकेशन की अनुमति दीजिए');
      return null;
    }
    return Geolocator.getCurrentPosition(
        locationSettings: const LocationSettings(accuracy: LocationAccuracy.high));
  }

  Future<void> _place() async {
    final items = [
      for (final e in _cart.entries)
        if (e.value > 0) {'product_id': e.key, 'qty': e.value}
    ];
    if (items.isEmpty) return;

    final addr = TextEditingController();
    final ok = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('डिलीवरी का पता'),
        content: TextField(
            controller: addr,
            decoration: const InputDecoration(labelText: 'फ़्लैट / लैंडमार्क')),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('रद्द')),
          FilledButton(
              style: FilledButton.styleFrom(minimumSize: const Size(120, 44)),
              onPressed: () => Navigator.pop(ctx, true),
              child: const Text('ऑर्डर करें')),
        ],
      ),
    );
    if (ok != true) return;

    setState(() => _placing = true);
    try {
      final pos = await _position();
      if (pos == null) return;
      final id = await supabase.rpc('place_order', params: {
        'p_items': items,
        'p_lat': pos.latitude,
        'p_lng': pos.longitude,
        'p_address': addr.text.trim().isEmpty ? 'GPS लोकेशन' : addr.text.trim(),
      });
      if (!mounted) return;
      setState(_cart.clear);
      await Navigator.of(context).push(MaterialPageRoute(
          builder: (_) => OrderTrackScreen(orderId: id as String)));
    } catch (e) {
      _msg('ऑर्डर नहीं हुआ: $e');
    } finally {
      if (mounted) setState(() => _placing = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Happy Food')),
      body: _loading
          ? const Center(child: CircularProgressIndicator())
          : _products.isEmpty
              ? const Center(child: Text('अभी कोई प्रोडक्ट नहीं है'))
              : RefreshIndicator(
                  onRefresh: _load,
                  child: ListView(
                    padding: const EdgeInsets.fromLTRB(16, 16, 16, 100),
                    children: [
                      for (final p in _products)
                        Card(
                          child: ListTile(
                            title: Text(p['name'],
                                style: const TextStyle(fontWeight: FontWeight.w700)),
                            subtitle: Text('₹${p['price']}',
                                style: const TextStyle(color: gold)),
                            trailing: Row(mainAxisSize: MainAxisSize.min, children: [
                              IconButton(
                                  icon: const Icon(Icons.remove_circle_outline),
                                  onPressed: (_cart[p['id']] ?? 0) > 0
                                      ? () => setState(
                                          () => _cart[p['id']] = _cart[p['id']]! - 1)
                                      : null),
                              Text('${_cart[p['id']] ?? 0}',
                                  style: const TextStyle(
                                      fontSize: 18, fontWeight: FontWeight.w800)),
                              IconButton(
                                  icon: const Icon(Icons.add_circle_outline),
                                  onPressed: () => setState(() =>
                                      _cart[p['id']] = (_cart[p['id']] ?? 0) + 1)),
                            ]),
                          ),
                        ),
                    ],
                  ),
                ),
      bottomNavigationBar: _total > 0
          ? SafeArea(
              child: Padding(
                padding: const EdgeInsets.all(16),
                child: FilledButton(
                  onPressed: _placing ? null : _place,
                  child: _placing
                      ? const SizedBox(
                          height: 22,
                          width: 22,
                          child: CircularProgressIndicator(strokeWidth: 2))
                      : Text('ऑर्डर करें  •  ₹$_total'),
                ),
              ),
            )
          : null,
    );
  }
}

class _OrderTrackScreenState extends State<OrderTrackScreen> {
  StreamSubscription<List<Map<String, dynamic>>>? _sub;
  Map<String, dynamic>? _order;
  String? _otp;

  @override
  void initState() {
    super.initState();
    _sub = supabase
        .from('orders')
        .stream(primaryKey: ['id'])
        .eq('id', widget.orderId)
        .listen((rows) async {
      if (rows.isEmpty) return;
      final o = rows.first;
      if ((o['status'] == 'accepted' || o['status'] == 'on_the_way') && _otp == null) {
        try {
          final r = await supabase.rpc('my_order_otp', params: {'p_order_id': widget.orderId});
          _otp = r as String?;
        } catch (_) {}
      }
      if (mounted) setState(() => _order = o);
    });
  }

  @override
  void dispose() {
    _sub?.cancel();
    super.dispose();
  }

  Future<void> _cancel() async {
    try {
      await supabase.rpc('cancel_order', params: {'p_order_id': widget.orderId});
    } catch (_) {}
  }

    final status = _order?['status'] as String?;

class _LoginScreenState extends State<LoginScreen> {
  final _email = TextEditingController();
  final _password = TextEditingController();
  bool _busy = false;
  bool _signUp = false;

  Future<void> _submit() async {
    final email = _email.text.trim();
    final password = _password.text;
    if (email.isEmpty || password.length < 6) {
      _msg('ईमेल डालिए और कम से कम 6 अक्षर का पासवर्ड रखिए');
      return;
    }
    setState(() => _busy = true);
    try {
      if (_signUp) {
        final res = await supabase.auth.signUp(email: email, password: password);
        if (res.session == null && mounted) {
          _msg('अकाउंट बन गया। ईमेल की पुष्टि करनी हो तो मेल चेक करें, फिर लॉगिन करें।');
          setState(() => _signUp = false);
        }
      } else {
        await supabase.auth.signInWithPassword(email: email, password: password);
      }
    } on AuthException catch (e) {
      _msg(e.message);
    } catch (e) {
      _msg('कुछ गड़बड़ हुई: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(m)));

class RiderOnlineTab extends StatefulWidget {
  const RiderOnlineTab({super.key});

  @override
  State<RiderOnlineTab> createState() => _RiderOnlineTabState();
}

class _RiderOnlineTabState extends State<RiderOnlineTab> {
  bool _online = false;
  bool _approved = false;
  bool _loading = true;

  Timer? _gps;
  Timer? _tick;
  StreamSubscription<List<Map<String, dynamic>>>? _offerSub;
  StreamSubscription<List<Map<String, dynamic>>>? _orderSub;

  List<Map<String, dynamic>> _offers = [];
  final Map<String, Map<String, dynamic>> _details = {};
  Map<String, dynamic>? _order;
  List<Map<String, dynamic>> _orderItems = [];

  String get _uid => supabase.auth.currentUser!.id;

  @override
  void initState() {
    super.initState();
    _load();
    _tick = Timer.periodic(const Duration(seconds: 1), (_) {
      if (mounted) setState(() {});
    });

    _offerSub = supabase
        .from('order_offers')
        .stream(primaryKey: ['id'])
        .eq('rider_id', _uid)
        .listen((rows) async {
      final pending = rows.where((r) => r['status'] == 'pending').toList();
      for (final r in pending) {
        final id = r['id'] as String;
        if (!_details.containsKey(id)) {
          try {
            final d = await supabase.rpc('offer_details', params: {'p_offer_id': id});
            if (d is Map) _details[id] = Map<String, dynamic>.from(d);
          } catch (_) {}
        }
      }
      if (mounted) setState(() => _offers = pending);
    });

    _orderSub = supabase
        .from('orders')
        .stream(primaryKey: ['id'])
        .eq('rider_id', _uid)
        .listen((rows) async {
      final active = rows
          .where((r) => r['status'] == 'accepted' || r['status'] == 'on_the_way')
          .toList();
      final cur = active.isEmpty ? null : active.first;
      if (cur != null && cur['id'] != _order?['id']) {
        final items = await supabase
            .from('order_items')
            .select('qty, products(name)')
            .eq('order_id', cur['id']);
        _orderItems = List<Map<String, dynamic>>.from(items);
      }
      if (mounted) setState(() => _order = cur);
    });
  }

  @override
  void dispose() {
    _gps?.cancel();
    _tick?.cancel();
    _offerSub?.cancel();
    _orderSub?.cancel();
    super.dispose();
  }

  void _msg(String m) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(m)));
  }

  Future<void> _load() async {
    try {
      final r = await supabase
          .from('riders')
          .select('is_online, approved')
          .eq('id', _uid)
          .maybeSingle();
      if (!mounted) return;
      setState(() {
        _online = r?['is_online'] == true;
        _approved = r?['approved'] == true;
      });
      if (_online) _startGps();
    } catch (e) {
      _msg('लोड नहीं हुआ: $e');
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  Future<bool> _ensureLocation() async {
    if (!await Geolocator.isLocationServiceEnabled()) {
      _msg('फ़ोन की लोकेशन (GPS) चालू कीजिए');
      return false;
    }
    var p = await Geolocator.checkPermission();
    if (p == LocationPermission.denied) p = await Geolocator.requestPermission();
    if (p == LocationPermission.denied || p == LocationPermission.deniedForever) {
      _msg('लोकेशन की अनुमति दीजिए');
      return false;
    }
    return true;
  }

  Future<void> _pushLocation() async {
    try {
      final pos = await Geolocator.getCurrentPosition(
          locationSettings: const LocationSettings(accuracy: LocationAccuracy.high));
      await supabase.rpc('update_rider_location',
          params: {'p_lat': pos.latitude, 'p_lng': pos.longitude});
    } catch (_) {}
  }

  void _startGps() {
    _gps?.cancel();
    _pushLocation();
    _gps = Timer.periodic(const Duration(seconds: 10), (_) => _pushLocation());
  }

  Future<void> _setOnline(bool v) async {
    if (v && !await _ensureLocation()) return;
    try {
      await supabase.from('riders').update({'is_online': v}).eq('id', _uid);
      setState(() => _online = v);
      if (v) {
        _startGps();
      } else {

  }

  Future<void> _load() async {
    try {
      final uid = supabase.auth.currentUser!.id;
      final p = await supabase
          .from('products')
          .select('id, name, price')
          .eq('is_active', true)
          .order('name');
      final s = await supabase
          .from('rider_stock')
          .select('product_id, qty, reserved')
          .eq('rider_id', uid);
      if (!mounted) return;
      setState(() {
        _products = List<Map<String, dynamic>>.from(p);
        _stock
          ..clear()
          ..addEntries(List<Map<String, dynamic>>.from(s)
              .map((r) => MapEntry(r['product_id'] as String, r)));
      });
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context)
            .showSnackBar(SnackBar(content: Text('लोड नहीं हुआ: $e')));
      }
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  Future<void> _set(String productId, int qty) async {
    try {
      await supabase.rpc('set_stock', params: {'p_product_id': productId, 'p_qty': qty});
      await _load();
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context)
            .showSnackBar(SnackBar(content: Text('गड़बड़: $e')));
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    if (_loading) return const Center(child: CircularProgressIndicator());
    if (_products.isEmpty) {
      return const Center(
          child: Padding(
              padding: EdgeInsets.all(24),

            Builder(builder: (_) {

/// Animated launch screen: glowing logo, orbiting food, rotating gold ring.
class SplashScreen extends StatefulWidget {
  const SplashScreen({super.key, required this.next});

  /// Screen to show after the splash (login or role picker).
  final Widget next;

  @override
  State<SplashScreen> createState() => _SplashScreenState();
}

class _SplashScreenState extends State<SplashScreen>
    with TickerProviderStateMixin {
  late final AnimationController _intro =
      AnimationController(vsync: this, duration: const Duration(milliseconds: 1600))
        ..forward();
  late final AnimationController _loop =
      AnimationController(vsync: this, duration: const Duration(seconds: 6))
        ..repeat();

  static const _food = ['🥟', '🍛', '🍢', '🍕', '🥘', '🍔'];

  @override
  void initState() {
    super.initState();
    Future.delayed(const Duration(milliseconds: 3000), _go);
  }

  void _go() {
    if (!mounted) return;
    Navigator.of(context).pushReplacement(PageRouteBuilder(
      transitionDuration: const Duration(milliseconds: 600),
      pageBuilder: (_, __, ___) => widget.next,
      transitionsBuilder: (_, a, __, child) =>
          FadeTransition(opacity: a, child: child),
    ));
  }

  @override
  void dispose() {
    _intro.dispose();
    _loop.dispose();
    super.dispose();
  }

  double _span(double a, double b) =>
      Curves.easeOutCubic.transform(((_intro.value - a) / (b - a)).clamp(0.0, 1.0).toDouble());

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFF0F0F15),
      body: AnimatedBuilder(
        animation: Listenable.merge([_intro, _loop]),
        builder: (context, _) {
          final pop = Curves.elasticOut.transform(_span(0.0, 0.55));
          final textIn = _span(0.45, 0.85);

            alignment: Alignment.center,
            children: [
              // soft golden glow
              Container(
                decoration: BoxDecoration(
                  gradient: RadialGradient(
                    center: const Alignment(0, -0.25),
                    radius: 0.95,
                    colors: [
                      gold.withOpacity(0.20 * _span(0, 0.6)),
                      Colors.transparent,
                    ],
                  ),

              ),
              Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  SizedBox(
                    width: 250,
                    height: 250,
                    child: Stack(
                      alignment: Alignment.center,
                      children: [
                        // rotating gold arc (radar feel)
                        Transform.rotate(
                          angle: t,
                          child: CustomPaint(
                            size: const Size(220, 220),
                            painter: _RingPainter(_span(0, 0.7)),
                          ),
                        ),

                                child: Text(_food[i],
                                    style: const TextStyle(fontSize: 26)),

class _RingPainter extends CustomPainter {
  _RingPainter(this.p);
  final double p;

  @override
  void paint(Canvas canvas, Size s) {
    final c = s.center(Offset.zero);
    final r = s.width / 2;
    canvas.drawCircle(
        c,
        r,
        Paint()
          ..style = PaintingStyle.stroke
          ..strokeWidth = 1.5
          ..color = gold.withOpacity(0.18 * p));
    final rect = Rect.fromCircle(center: c, radius: r);
    canvas.drawArc(
        rect,
        0,
        math.pi * 0.9 * p,
        false,
        Paint()
          ..style = PaintingStyle.stroke
          ..strokeCap = StrokeCap.round
          ..strokeWidth = 4
          ..shader = SweepGradient(
            colors: [gold.withOpacity(0), gold],
            stops: const [0, 1],
            transform: const GradientRotation(0),
          ).createShader(rect));
  }

  @override
  bool shouldRepaint(_RingPainter o) => o.p != p;
}
